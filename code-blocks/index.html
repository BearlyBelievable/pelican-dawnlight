import json


class ApplicationForm:
    """A form built from a list of fields."""

    def __init__(self, fields):
        self.fields = fields

    @classmethod
    def load(cls, path):
        """Read the fields from a JSON file."""
        # Open the file and parse it
        with open(path) as f:
            return cls(json.load(f))
